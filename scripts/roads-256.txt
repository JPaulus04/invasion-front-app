 function prepareRoad255(im,mask){
  if(typeof document==='undefined')return null;
  var cv=document.createElement('canvas');cv.width=256;cv.height=128;var ctx=cv.getContext('2d');if(!ctx)return null;
  ctx.drawImage(im,0,0,256,128);var px=ctx.getImageData(0,0,256,128),data=px.data,original=new Uint8ClampedArray(data);
  var ends=[];[[1,.55,0],[2,-.55,0],[4,0,.55],[8,0,-.55]].forEach(function(a){if(mask&a[0])ends.push([a[1],a[2]]);});
  var segments=[];
  if(ends.length===1)segments.push([0,0,ends[0][0],ends[0][1]]);
  for(var a=0;a<ends.length;a++)for(var b=a+1;b<ends.length;b++){
   var e=ends[a],f=ends[b],last=e;
   for(var step=1;step<=8;step++){
    var t=step/8,point=[(1-t)*(1-t)*e[0]+t*t*f[0],(1-t)*(1-t)*e[1]+t*t*f[1]];
    segments.push([last[0],last[1],point[0],point[1]]);last=point;
   }
  }
  for(var y=0;y<128;y++)for(var x=0;x<256;x++){
   var i=(y*256+x)*4,u=(x+.5-128)/256+(y+.5-64)/128,v=(y+.5-64)/128-(x+.5-128)/256,dist=10;
   for(var k=0;k<segments.length;k++){
    var q=segments[k],dx=q[2]-q[0],dy=q[3]-q[1],f=Math.max(0,Math.min(1,((u-q[0])*dx+(v-q[1])*dy)/(dx*dx+dy*dy||1)));
    var du=u-q[0]-f*dx,dv=v-q[1]-f*dy;dist=Math.min(dist,du*du+dv*dv);
   }
   dist=Math.sqrt(dist);var width=.14,alpha=Math.max(0,Math.min(1,(width+.009-dist)/.018));
   // Fill formerly transparent source corners from clean central dirt, never the matte.
   var j=i;
   if(original[i+3]<220||(original[i]>110&&original[i+2]>95&&original[i]>original[i+1]*1.3&&original[i+2]>original[i+1]*1.3))j=((61+y%6)*256+122+x%12)*4;
   var shade=1-Math.max(0,(dist-.095)/.05)*.13;
   for(var c=0;c<3;c++)data[i+c]=original[j+c]*shade;
   data[i+3]=Math.round(alpha*255);
  }
  ctx.putImageData(px,0,0);return cv;
 }
